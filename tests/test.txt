# ==========================================
# 1. BASIC ARITHMETIC & PRECEDENCE
# ==========================================
3+4*2 = 11
(3+4)*2 = 14
10/2 - 1 = 4
100 - 50 + 25 = 75
5 * 5 * 5 = 125

# ==========================================
# 2. DECIMALS AND FLOATS
# ==========================================
2.5 * 4 = 10
10 / 4 = 2.5
0.5 + 0.5 = 1
1.5 * 1.5 = 2.25

# ==========================================
# 3. NESTED PARENTHESES
# ==========================================
((2 + 3) * (4 + 1)) = 25
10 / (2 + 3) = 2
(1+(2+(3+4))) = 10

# ==========================================
# 4. UNARY MINUS (Negative Numbers)
# ==========================================
-5 = -5
-(3+4) = -7
3*-4 = -12
-10 + 5 = -5

# ==========================================
# 5. EXPONENTS (Right-Associative)
# ==========================================
2^8 = 256
2^3^2 = 512
3^2 + 4^2 = 25

# ==========================================
# 6. COMBINED: UNARY + EXPONENTS
# ==========================================
(-2)^4 = 16
(-2)^3 = -8
